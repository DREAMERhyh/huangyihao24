# P1 代码演进史（汇报用）

> **用途**：任务之间会相互覆盖代码（例如 `init/main.c` 被任务 3、4、5 连改三代，
> 任务 3 的形态在最终代码里已经看不到了）。本文件把**每个任务当时"能跑通"的
> 完整代码**从 git 历史中原样导出，汇报时按任务逐段讲解，不必在最终代码里翻找历史。
>
> **可靠性**：全部内容由脚本直接调用 `git show` 导出，与对应提交**逐字节一致**，
> 没有手工转录，不会抄错。
>
> **§8 是学习问答集**：把学习过程中提出的问题与答案按任务整理（含反汇编证据），
> 可直接当汇报的"预设答辩"材料用。

```bash

### 0.3 本文件是怎么生成的

本文件由下面这个脚本生成（把每个提交的快照导出为 Markdown）：

```bash
#!/bin/bash
emit_file() {   # $1=提交号  $2=文件路径
    hash="$1"; path="$2"
    printf '\n#### `%s`\n\n> 提交 `%s` 时的完整内容\n\n```\n' "$path" "$hash"
    git show "$hash:$path"
    printf '```\n'
}
emit_stat() {   # $1=提交号
    printf '\n**本步改动的文件清单**\n\n```\n'
    git show --stat --format="commit %h%n%s" "$1"
    printf '```\n'
}
# 然后按 "起点 / 任务1..5" 依次调用 emit_stat 与 emit_file
```

/* TODO: [p1-task4] assign your task_info_t somewhere in 'create_image' */
static void create_image(int nfiles, char *files[])
{
    int tasknum = nfiles - 2;
    int nbytes_kernel = 0;
    int phyaddr = 0;
    FILE *fp = NULL, *img = NULL;
    Elf64_Ehdr ehdr;
    Elf64_Phdr phdr;

    /* open the image file */
    img = fopen(IMAGE_FILE, "w");
    assert(img != NULL);

    /* for each input file */
    for (int fidx = 0; fidx < nfiles; ++fidx) {

        int taskidx = fidx - 2;

        /* open input file */
        fp = fopen(*files, "r");
        assert(fp != NULL);

        /* read ELF header */
        read_ehdr(&ehdr, fp);
        printf("0x%04lx: %s\n", ehdr.e_entry, *files);

        /* for each program header */
        for (int ph = 0; ph < ehdr.e_phnum; ph++) {

            /* read program header */
            read_phdr(&phdr, fp, ph, ehdr);

            if (phdr.p_type != PT_LOAD) continue;

            /* write segment to the image */
            write_segment(phdr, fp, img, &phyaddr);

            /* update nbytes_kernel */
            if (strcmp(*files, "main") == 0) {
                nbytes_kernel += get_filesz(phdr);
            }
        }

        /* write padding bytes */
        /**
         * TODO:
         * 1. [p1-task3] do padding so that the kernel and every app program
         *  occupies the same number of sectors
         * 2. [p1-task4] only padding bootblock is allowed!
         */
        if (strcmp(*files, "bootblock") == 0) {
            write_padding(img, &phyaddr, SECTOR_SIZE);
        }

        fclose(fp);
        files++;
    }
    write_img_info(nbytes_kernel, taskinfo, tasknum, img);

    fclose(img);
}

static void read_ehdr(Elf64_Ehdr * ehdr, FILE * fp)
{
    int ret;

    ret = fread(ehdr, sizeof(*ehdr), 1, fp);
    assert(ret == 1);
    assert(ehdr->e_ident[EI_MAG1] == 'E');
    assert(ehdr->e_ident[EI_MAG2] == 'L');
    assert(ehdr->e_ident[EI_MAG3] == 'F');
}

static void read_phdr(Elf64_Phdr * phdr, FILE * fp, int ph,
                      Elf64_Ehdr ehdr)
{
    int ret;

    fseek(fp, ehdr.e_phoff + ph * ehdr.e_phentsize, SEEK_SET);
    ret = fread(phdr, sizeof(*phdr), 1, fp);
    assert(ret == 1);
    if (options.extended == 1) {
        printf("\tsegment %d\n", ph);
        printf("\t\toffset 0x%04lx", phdr->p_offset);
        printf("\t\tvaddr 0x%04lx\n", phdr->p_vaddr);
        printf("\t\tfilesz 0x%04lx", phdr->p_filesz);
        printf("\t\tmemsz 0x%04lx\n", phdr->p_memsz);
    }
}

static uint64_t get_entrypoint(Elf64_Ehdr ehdr)
{
    return ehdr.e_entry;
}

static uint32_t get_filesz(Elf64_Phdr phdr)
{
    return phdr.p_filesz;
}

static uint32_t get_memsz(Elf64_Phdr phdr)
{
    return phdr.p_memsz;
}

static void write_segment(Elf64_Phdr phdr, FILE *fp, FILE *img, int *phyaddr)
{
    if (phdr.p_memsz != 0 && phdr.p_type == PT_LOAD) {
        /* write the segment itself */
        /* NOTE: expansion of .bss should be done by kernel or runtime env! */
        if (options.extended == 1) {
            printf("\t\twriting 0x%04lx bytes\n", phdr.p_filesz);
        }
        fseek(fp, phdr.p_offset, SEEK_SET);
        while (phdr.p_filesz-- > 0) {
            fputc(fgetc(fp), img);
            (*phyaddr)++;
        }
    }
}

static void write_padding(FILE *img, int *phyaddr, int new_phyaddr)
{
    if (options.extended == 1 && *phyaddr < new_phyaddr) {
        printf("\t\twrite 0x%04x bytes for padding\n", new_phyaddr - *phyaddr);
    }

    while (*phyaddr < new_phyaddr) {
        fputc(0, img);
        (*phyaddr)++;
    }
}

#include <asm.h>

// 用户程序的栈：所有 app 共享一个栈，栈顶 0x50400000，向下生长
//   内核栈独享 0x50400000 ~ 0x50500000（KERNEL_STACK = 0x50500000 向下生长），
//   两者互不交叠；app 栈向下到 0x50200000 仍有约 2MB 空间
.equ APP_STACK_TOP, 0x50400000

.section ".entry_function","ax"
ENTRY(_start)

    /* 任务3：为用户程序准备 C 运行环境
     *
     * 先保存内核现场。必须放进 .data 里的固定槽位，原因是：
     *   - 不能借 a0/a1 等临时寄存器：app 的 main 会随意使用它们，一冲就没
     *   - 不能借 s0-s11：内核调进来时 main() 里可能还有活着的 s 寄存器，
     *     而被调用者（也就是本函数）按 ABI 绝不能破坏它们
     * 因此本函数全程只使用 t / a 寄存器 */
    la    t2, saved_ra
    sd    ra, 0(t2)
    la    t3, saved_sp
    sd    sp, 0(t3)

    /* app 的栈：所有 app 共享，栈顶 0x50400000，向下生长 */
    li    sp, APP_STACK_TOP

    /* 清零 app 的 BSS
     *   边界取自本 app 自己 ELF 的 __bss_start / __BSS_END__
     *   （由共用链接脚本 riscv.lds 在该 app 自己的布局中赋值）
     *   注意必须按字节清（sb）：__bss_start 可能不是 8 字节对齐，
     *   例如 auipc 的 __bss_start = 0x52010284 且 bss 只有 4 字节，
     *   用 sd 会越界写坏后面的内存 */
    la    t0, __bss_start
    la    t1, __BSS_END__
clear_loop:
    bge   t0, t1, clear_done
    sb    zero, 0(t0)
    addi  t0, t0, 1
    j     clear_loop
clear_done:

    /* 任务3：进入 app 的 main 函数 */
    call  main

    /* 任务3：app 返回后跳回内核（恢复刚才保存的内核现场） */
    la    t2, saved_sp
    ld    sp, 0(t2)
    la    t3, saved_ra
    ld    ra, 0(t3)
    jr    ra

    /************************************************************/
	/* Do not touch this comment. Reserved for future projects. */
	/************************************************************/
// while(1) loop, unreachable here
loop:
    wfi
    j loop

END(_start)

.data
/* 内核现场保存槽位。必须放 .data 而不是 .bss：
 * 否则上面那段 BSS 清零循环会把自己刚存进去的值抹掉 */
saved_ra: .zero 8
saved_sp: .zero 8

    // 入口地址为什么可以直接用窗口基址，而不必从镜像里读 e_entry：
    //   1) riscv.lds 里 ENTRY(_start)，且 .text 段的第一个输入段是 *(.entry_function)
    //   2) crt0.S 的 _start 定义在 .entry_function 段，所以排在 app 代码之前
    //   3) Makefile 用 -Wl,--defsym=TEXT_START=<窗口基址> 给每个 app 单独赋值
    //   因此每个 app 的 _start 恰好落在自己的窗口基址上。实测四个 app 的
    //   e_entry 与窗口基址完全一致：2048=0x52000000 auipc=0x52010000
    //   bss=0x52020000 data=0x52030000。
    //   Task 4 改成按名加载时，e_entry 会存进 task_info_t，届时以镜像里的值为准。
    uint64_t entry = TASK_MEM_BASE + (uint64_t)taskid * TASK_SIZE;
    uint32_t block = 1 + KERNEL_FIXED_SECTORS + (uint32_t)taskid * APP_FIXED_SECTORS;

    // 一次读满该 app 的分配扇区数；镜像里的 padding 全是 0，
    // 所以窗口内不会被残留数据污染
    if (bios_sd_read((unsigned)entry, APP_FIXED_SECTORS, block) != 0)
        return 0;   // 读取失败，用 0 作为错误哨兵（合法入口恒 >= 0x52000000）

    return entry;
}

// os size location (os_size could be sectors [p1-task3] or bytes [p1-task4])
.equ os_size_loc, 0x502001fc

// kernel address (move kernel to here, and jmp here to start kernel)
.equ kernel, 0x50201000

// BIOS function entry (jump here to use BIOS APIs)
.equ bios_func_entry, 0x50150000

.text
.global main

main:
	// fence on all memory and I/O
	fence

	// 任务1：调用 BIOS 的 bios_putstr 打印 bootblock 标志串
	//   调用约定：a0 = 字符串首地址，a7 = 功能号，然后跳转到 bios_func_entry
	//   0x50150000 处是一条 ecall + ret，ret 依赖 ra 返回，
	//   用 call(即 jal) 可让汇编器自动填好返回地址 ra
	la    a0, msg
	li    a7, BIOS_PUTSTR
	call  bios_func_entry


	// 任务2：调用 BIOS 的 bios_sd_read 把内核从 SD 卡读入内存
	//   内核占用的扇区数由 createimage 写在镜像偏移 0x1fc 处
	//   （即内存 0x502001fc，位于 bootblock 自己扇区的尾部），这里用 lh 读出
	//   镜像布局：第 0 扇区 = bootblock，第 1 扇区起 = 内核
	//   加载目标：0x50201000，每个扇区 512 字节
	li    t3, os_size_loc       // t3 = 0x502001fc（扇区数字段地址）
	lh    t2, 0(t3)             // t2 = 内核占用的扇区数
	li    t0, 1                 // t0 = 起始扇区号（SD 卡第 1 扇区）
	li    t1, kernel            // t1 = 目标内存地址

read_loop:
	beqz  t2, read_done         // 扇区数为 0 则不读（防御性判断）

	move  a0, t1                // a0 = 本次读取的目标内存地址
	li    a1, 1                 // a1 = 每次读 1 个扇区
	move  a2, t0                // a2 = 本次读取的扇区号
	li    a7, BIOS_SDREAD       // a7 = 11
	call  bios_func_entry

	addi  t1, t1, 512           // 下一个扇区的目标地址
	addi  t0, t0, 1             // 下一个扇区号
	addi  t2, t2, -1            // 剩余扇区数减一
	j     read_loop

read_done:

	// 任务4：把镜像布局参数传给内核（替代内核直接读 0x50200xxx）：
	//   布局：扇区 0 = bootblock，1..1+K-1 = 内核，1+K = AppInfo 区，
	//         之后 = 各 app 紧凑排列。K 是上面从 0x1fc 读出的内核扇区数。
	//   传参约定（head.S 会保存，见 task.h）：
	//     a0 = AppInfo 区起始扇区号 = 1 + K
	//          （t0 在循环结束时正好停在 1+K，直接复用）
	//     a1 = 用户程序数目（从本扇区偏移 0x1f8 读出）
	//   这依赖 U-Boot `virtio read 50200000 0 2` 预读前 2 扇区到 0x50200000：
	//   0x502001fc/0x1f8 才在内存里可读（见 log/task3 前置项 3 的隐藏依赖说明）。
	move  a0, t0               // a0 = 1 + K（AppInfo 起始扇区）
	li    t3, os_size_loc - 4  // t3 = 0x502001f8（app 数目字段地址）
	lh    a1, 0(t3)            // a1 = 用户程序数目

	// 任务2：跳转到内核入口，启动 UCAS-OS

static void init_jmptab(void)
{

static void init_task_info(void)
{
    // 任务4：从 AppInfo 区读出 task_info_t 数组，填进 tasks[]。
    //   位置与数量来自 bootblock 传参（head.S 已存进 boot_appinfo_sector /
    //   boot_appnum），不再直接读 0x50200xxx。
    //   AppInfo 必须先读到本地 512B 对齐缓冲中转，再拷 task_num*sizeof 字节
    //   进 tasks[]，不能整扇区直读 tasks[]（会越界覆写 tasks 后面的 .bss）。
    // NOTE: [p1-task3] 的 taskid 读取已废弃，改按名字加载（见 main 循环）。
    static uint8_t appinfo_buf[SECTOR_SIZE]
        __attribute__((aligned(SECTOR_SIZE)));

    task_num = (int)boot_appnum;

    if (task_num <= 0 || task_num > TASK_MAXNUM) {
        task_num = 0;
        return;
    }

    if (bios_sd_read((unsigned)(unsigned long)appinfo_buf, APPINFO_SECTORS,
                     (unsigned)boot_appinfo_sector) != 0) {
        task_num = 0;
        return;
    }

    memcpy((void *)tasks, (const void *)appinfo_buf,
           (size_t)task_num * sizeof(task_info_t));
}

// 任务4：读入一行由 \r 或 \n 结束的程序名。
//   支持退格（BS=0x08 / DEL=0x7f）；名字截断到 TASK_NAME_LEN-1；
//   若某行超长则"整个名字"判非法，清空到行尾后返回 0。
//   \r 与 \n 都立即终止；后续的 \r 或 \n（如 \r\n 的另一半）会以空行被
//   瞬间消费掉，由调用方忽略空行，不会卡住。
static int readline(char *buf, int cap)
{
    int len = 0;

    buf[0] = '\0';
    while (1)
    {
        int ch = bios_getchar();

        if (ch == -1)
            continue;   // 无输入跳过

        if (ch == '\n' || ch == '\r') {   // 双终止符
            buf[len] = '\0';
            return len;
        }

        if (ch == '\b' || ch == 0x7f) {   // 退格：BS 或 DEL
            if (len > 0) {
                len--;
                bios_putchar('\b');        // 回显擦除
                bios_putchar(' ');
                bios_putchar('\b');
            }
            continue;
        }

        if (ch < ' ' || ch > '~')         // 不可打印字符忽略
            continue;

        if (len >= cap - 1) {              // 名字超长：本行作废
            while (1) {
                while ((ch = bios_getchar()) == -1)
                    ;
                if (ch == '\n' || ch == '\r')
                    break;
            }
            buf[0] = '\0';
            return 0;
        }

        bios_putchar(ch);                  // 回显
        buf[len++] = (char)ch;
    }
}

    int check = bss_check();

    // Init jump table provided by kernel and bios(ΦωΦ)
    init_jmptab();

    // Init task information from bootblock-passed AppInfo (〃'▽'〃)
    init_task_info();

    // Output 'Hello OS!', bss check result and OS version
    char output_str[] = "bss check: _ version: _\n\r";
    char output_val[2] = {0};
    int i, output_val_pos = 0;

    output_val[0] = check ? 't' : 'f';
    output_val[1] = version + '0';
    for (i = 0; i < sizeof(output_str); ++i)
    {
        buf[i] = output_str[i];
        if (buf[i] == '_')
        {
            buf[i] = output_val[output_val_pos++];
        }
    }

    bios_putstr("Hello OS!\n\r");
    bios_putstr(buf);

    // 任务4：列出可用的任务名，方便按名加载
    bios_putstr("Available tasks: ");
    for (int i = 0; i < task_num; ++i) {
        bios_putstr(tasks[i].name);
        if (i + 1 < task_num)
            bios_putstr(", ");
    }
    bios_putstr("\n\r");

    // 任务4：交互式读入程序名，在 AppInfo 里匹配、加载并执行
    //   未知名/空行/超长行都提示错误并回到输入循环，不得跑飞
    while (1)
    {
        char name[TASK_NAME_LEN];

        bios_putstr("Please input a task name: ");

        int len = readline(name, TASK_NAME_LEN);

        if (len <= 0) {
            bios_putstr("Error: invalid input!\n\r");
            continue;
        }

        uint64_t entry = load_task_img(name);

        if (entry == 0)
        {
            bios_putstr("Error: unknown task or load failed!\n\r");
            continue;
        }

        // 跳转到 app 入口；app 的 crt0 会在 app main 返回后跳回内核
        //
        // 关于返回地址的说明（勘误后）：GCC 对函数指针调用本就会生成
        //   jalr ra, rs（rd=x1，返回地址进 ra）；objdump 会把 rd=ra 的
        //   jalr 简写成 `jalr a0` 这类省略形式（见 log/task3-工作日志.md
        //   "jalr 根因复核"），早期据此误判为"返回地址进 a0"。
        // 这里仍显式先设 ra 再间接跳转，属于双保险：行为与 GCC 隐式生成
        //   完全等价，且把"返回地址落在哪"写清楚，便于阅读与维护。
        asm volatile(
            "la   ra, 1f\n\t"
            "jalr a0, %0\n\t"
            "1:\n\t"
            :
            : "r"(entry)
            : "ra", "a0", "memory"
        );
    }

    return 0;
}

#include <assert.h>
#include <elf.h>
#include <errno.h>
#include <stdarg.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define IMAGE_FILE "./image"
#define ARGS "[--extended] [--vm] <bootblock> <executable-file> ..."

#define SECTOR_SIZE 512
#define BOOT_LOADER_SIG_OFFSET 0x1fe
#define OS_SIZE_LOC (BOOT_LOADER_SIG_OFFSET - 2)
#define BOOT_LOADER_SIG_1 0x55
#define BOOT_LOADER_SIG_2 0xaa

#define NBYTES2SEC(nbytes) (((nbytes) / SECTOR_SIZE) + ((nbytes) % SECTOR_SIZE != 0))

/* 任务4 的 task_info_t，与 include/os/task.h 完全一致（必须保持同步，
 * 大小 16+4+4+4 = 28 字节，4 字节对齐；TASK_MAXNUM * 28 = 448 < 512，
 * 所以 AppInfo 区固定 = 1 扇区）。 */
#define TASK_NAME_LEN 16
#define TASK_MAXNUM   16
typedef struct {
    char     name[TASK_NAME_LEN];
    uint32_t start_sector;
    uint32_t sector_num;
    uint32_t vaddr_entry;
} task_info_t;
static task_info_t taskinfo[TASK_MAXNUM];

/* 镜像信息区（位于第 0 扇区尾部）。用户程序数目 @0x1f8（避开官方签名 @0x1fe）。 */
#define IMG_APP_NUM_LOC (OS_SIZE_LOC - 4)

/* 任务4 镜像布局（紧凑排列，无固定空泡）：
 *
 *   扇区 0                bootblock
 *   扇区 1 .. K           内核，K = NBYTES2SEC(内核实际大小)
 *   扇区 K+1              AppInfo 区（task_info_t 数组，固定 1 扇区）
 *   扇区 K+2 ..           各 app 按实际大小紧凑排列（app i 紧贴 app i-1）
 *
 * bootblock 从第 0 扇区尾部读出 K 与 app 数目，把 AppInfo 扇区号(1+K)与
 * app 数目经寄存器传给内核（见 arch/riscv/boot/bootblock.S、head.S）。
 */
#define APPINFO_SEC_START(k) ((k) + 1)

/* structure to store command line options */
static struct {

/* prototypes of local functions */

    options.vm = 0;
    options.extended = 0;
    while ((argc > 1) && (argv[1][0] == '-') && (argv[1][1] == '-')) {

/* 预扫描：返回 ELF 中所有 PT_LOAD 段的 p_filesz 之和，并输出入口地址 */
static uint32_t scan_elf(const char *path, uint64_t *entry_out)
{
    FILE *fp = fopen(path, "r");
    Elf64_Ehdr ehdr;
    Elf64_Phdr phdr;
    uint32_t filesz_sum = 0;

    assert(fp != NULL);
    read_ehdr(&ehdr, fp);
    *entry_out = ehdr.e_entry;

    for (int ph = 0; ph < ehdr.e_phnum; ph++) {
        read_phdr(&phdr, fp, ph, ehdr);
        if (phdr.p_type != PT_LOAD)
            continue;
        filesz_sum += get_filesz(phdr);
    }
    fclose(fp);
    return filesz_sum;
}

static void create_image(int nfiles, char *files[])
{
    int tasknum = nfiles - 2;
    int phyaddr = 0;
    FILE *img = NULL;
    Elf64_Ehdr ehdr;
    Elf64_Phdr phdr;

    /* 每个文件的布局信息 */
    uint32_t used_bytes[TASK_MAXNUM + 2];   /* 顺序：bootblock, main, app... */
    uint64_t entry_addr[TASK_MAXNUM + 2];

    /* 预扫描一遍文件，得到每个文件实际会写进镜像的字节数 */
    for (int fidx = 0; fidx < nfiles; ++fidx)
        used_bytes[fidx] = scan_elf(files[fidx], &entry_addr[fidx]);

    /* ---- 布局计算（全部按扇区） ---- */
    int boot_sec        = 0;
    int kernel_sec      = 1;
    int kernel_sectors  = NBYTES2SEC(used_bytes[1]);            /* 内核 */
    int appinfo_sec     = APPINFO_SEC_START(kernel_sectors);    /* 1+K */
    int appinfo_sectors = NBYTES2SEC(TASK_MAXNUM * sizeof(task_info_t)); /* 1 */
    int app_sec[TASK_MAXNUM];
    int cur = appinfo_sec + appinfo_sectors;

    for (int i = 0; i < tasknum; ++i) {
        app_sec[i] = cur;
        cur += NBYTES2SEC(used_bytes[2 + i]);
    }
    int total_sectors = cur;

    printf("===== 镜像布局（紧凑排列） =====\n");

    /* ---- 写镜像 ---- */
    img = fopen(IMAGE_FILE, "w");
    assert(img != NULL);

    /* 第 0 扇区：bootblock */
    FILE *fp = fopen(files[0], "r");
    assert(fp != NULL);
    read_ehdr(&ehdr, fp);
    printf("0x%04lx: %-8s sec %2d .. %2d  (%d bytes, 1 sector)\n",
           ehdr.e_entry, files[0], boot_sec, boot_sec,
           used_bytes[0]);
    for (int ph = 0; ph < ehdr.e_phnum; ph++) {
        read_phdr(&phdr, fp, ph, ehdr);
        if (phdr.p_type != PT_LOAD)
            continue;
        write_segment(phdr, fp, img, &phyaddr);
    }
    fclose(fp);
    if (phyaddr > SECTOR_SIZE) {
        error("bootblock (%d bytes) exceeds one sector (%d bytes)\n",
              phyaddr, SECTOR_SIZE);
    }
    write_padding(img, &phyaddr, SECTOR_SIZE);

    /* 扇区 1..K：内核 */
    fp = fopen(files[1], "r");
    assert(fp != NULL);
    read_ehdr(&ehdr, fp);
    printf("0x%04lx: %-8s sec %2d .. %2d  (%d bytes, %d sectors)\n",
           ehdr.e_entry, files[1], kernel_sec,
           kernel_sec + kernel_sectors - 1, used_bytes[1],
           kernel_sectors);
    for (int ph = 0; ph < ehdr.e_phnum; ph++) {
        read_phdr(&phdr, fp, ph, ehdr);
        if (phdr.p_type != PT_LOAD)
            continue;
        write_segment(phdr, fp, img, &phyaddr);
    }
    fclose(fp);
    write_padding(img, &phyaddr, (1 + kernel_sectors) * SECTOR_SIZE);

    /* 扇区 K+1：AppInfo 区（task_info_t 数组，槽位清零） */
    printf("-APPINFO--- %-8s sec %2d .. %2d  (%d bytes, %d sectors)\n",
           "appinfo", appinfo_sec,
           appinfo_sec + appinfo_sectors - 1,
           (int)(TASK_MAXNUM * sizeof(task_info_t)), appinfo_sectors);
    for (int i = 0; i < tasknum; ++i) {
        /* 文件名可能很长：超过槽位就报错，绝不截断 */
        if (strlen(files[2 + i]) >= TASK_NAME_LEN) {
            error("task name '%s' too long (>= %d)\n", files[2 + i],
                  TASK_NAME_LEN);
        }
        strcpy(taskinfo[i].name, files[2 + i]);
        taskinfo[i].start_sector = app_sec[i];
        taskinfo[i].sector_num   = NBYTES2SEC(used_bytes[2 + i]);
        taskinfo[i].vaddr_entry  = (uint32_t)entry_addr[2 + i];
    }
    /* 剩余槽位清零（未用槽位的 name[0]=='\0'，内核靠它识别空槽） */
    for (int i = tasknum; i < TASK_MAXNUM; ++i)
        memset(&taskinfo[i], 0, sizeof(task_info_t));
    write_appinfo(tasknum, appinfo_sec * SECTOR_SIZE, img);
    /* fwrite 不经过 phyaddr 计数：同步一次，再把该扇区尾部(448B 之后)清零 */
    phyaddr = appinfo_sec * SECTOR_SIZE +
              (int)(TASK_MAXNUM * sizeof(task_info_t));
    write_padding(img, &phyaddr, (appinfo_sec + appinfo_sectors) * SECTOR_SIZE);

    /* 扇区 K+2 起：各 app 紧凑排列（无固定空泡） */
    for (int i = 0; i < tasknum; ++i) {
        fp = fopen(files[2 + i], "r");
        assert(fp != NULL);
        read_ehdr(&ehdr, fp);
        printf("0x%04lx: %-8s sec %2d .. %2d  (%d bytes, %d sectors)\n",
               ehdr.e_entry, files[2 + i], app_sec[i],
               app_sec[i] + taskinfo[i].sector_num - 1,
               used_bytes[2 + i], taskinfo[i].sector_num);
        for (int ph = 0; ph < ehdr.e_phnum; ph++) {
            read_phdr(&phdr, fp, ph, ehdr);
            if (phdr.p_type != PT_LOAD)
                continue;
            write_segment(phdr, fp, img, &phyaddr);
        }
        fclose(fp);
        write_padding(img, &phyaddr, (app_sec[i] + taskinfo[i].sector_num) *
                                     SECTOR_SIZE);
    }

    /* 布局自检：按公式核算总扇区数，与文件实际大小必须一致 */
    int actual_total = phyaddr / SECTOR_SIZE;
    if (actual_total != total_sectors) {
        error("layout mismatch: computed %d sectors, wrote %d\n",
              total_sectors, actual_total);
    }

    /* ---- 镜像信息字段 ---- */
    write_img_info((uint16_t)kernel_sectors, (short)tasknum, img);

    printf("\n===== 布局核算 =====");
    printf("\n[layout] total sectors : %d = 1(boot) + %d(kernel) + %d(appinfo)"
           " + %d(apps)\n",
           total_sectors, kernel_sectors, appinfo_sectors,
           total_sectors - 1 - kernel_sectors - appinfo_sectors);
    printf("[layout] appinfo sector : %d (= 1 + kernel_sectors)\n",
           appinfo_sec);
    printf("[layout] app#1 sector   : %d (= 1 + kernel_sectors + %d)"
           "  (无固定空泡)\n",
           tasknum > 0 ? app_sec[0] : -1, appinfo_sectors);

    fclose(img);
}

static void read_ehdr(Elf64_Ehdr * ehdr, FILE * fp)
{
    int ret;

    ret = fread(ehdr, sizeof(*ehdr), 1, fp);
    assert(ret == 1);
    assert(ehdr->e_ident[EI_MAG1] == 'E');
    assert(ehdr->e_ident[EI_MAG2] == 'L');
    assert(ehdr->e_ident[EI_MAG3] == 'F');
}

static void read_phdr(Elf64_Phdr * phdr, FILE * fp, int ph,
                      Elf64_Ehdr ehdr)
{
    int ret;

    fseek(fp, ehdr.e_phoff + ph * ehdr.e_phentsize, SEEK_SET);
    ret = fread(phdr, sizeof(*phdr), 1, fp);
    assert(ret == 1);
    if (options.extended == 1) {
        printf("\tsegment %d\n", ph);
        printf("\t\toffset 0x%04lx", phdr->p_offset);
        printf("\t\tvaddr 0x%04lx\n", phdr->p_vaddr);
        printf("\t\tfilesz 0x%04lx", phdr->p_filesz);
        printf("\t\tmemsz 0x%04lx\n", phdr->p_memsz);
    }
}

static uint32_t get_filesz(Elf64_Phdr phdr)
{
    return phdr.p_filesz;
}

static void write_segment(Elf64_Phdr phdr, FILE *fp, FILE *img, int *phyaddr)
{
    if (phdr.p_memsz != 0 && phdr.p_type == PT_LOAD) {
        /* write the segment itself */
        /* NOTE: expansion of .bss should be done by kernel or runtime env! */
        if (options.extended == 1) {
            printf("\t\twriting 0x%04lx bytes\n", phdr.p_filesz);
        }
        fseek(fp, phdr.p_offset, SEEK_SET);
        while (phdr.p_filesz-- > 0) {
            fputc(fgetc(fp), img);
            (*phyaddr)++;
        }
    }
}

static void write_padding(FILE *img, int *phyaddr, int new_phyaddr)
{
    if (options.extended == 1 && *phyaddr < new_phyaddr) {
        printf("\t\twrite 0x%04x bytes for padding\n", new_phyaddr - *phyaddr);
    }

    while (*phyaddr < new_phyaddr) {
        fputc(0, img);
        (*phyaddr)++;
    }
}

/* 任务4：把 task_info_t 数组写到 AppInfo 扇区。
 *   整个数组（TASK_MAXNUM 个槽位）都写，未用的槽位在调用前已清 0。 */
static void write_appinfo(int appnum, int appinfo_offset, FILE *img)
{
    fseek(img, appinfo_offset, SEEK_SET);
    fwrite(taskinfo, sizeof(task_info_t), TASK_MAXNUM, img);
    if (appnum > 0) {
        printf("[appinfo] %d task(s) @ 0x%04x\n", appnum, appinfo_offset);
    }
}

static void write_img_info(uint16_t ksectors, short tasknum, FILE * img)

    /* 任务4：内核扇区数写"实际占用"（紧凑布局下 bootblock 精确读内核），

#include <asm/biosdef.h>

// os size location (os_size could be sectors [p1-task3] or bytes [p1-task4])
.equ os_size_loc, 0x502001fc

// kernel address (move kernel to here, and jmp here to start kernel)
.equ kernel, 0x50201000

// BIOS function entry (jump here to use BIOS APIs)
.equ bios_func_entry, 0x50150000

.text
.global main

main:
	// fence on all memory and I/O
	fence

	// 任务1：调用 BIOS 的 bios_putstr 打印 bootblock 标志串
	//   调用约定：a0 = 字符串首地址，a7 = 功能号，然后跳转到 bios_func_entry
	//   0x50150000 处是一条 ecall + ret，ret 依赖 ra 返回，
	//   用 call(即 jal) 可让汇编器自动填好返回地址 ra
	la    a0, msg
	li    a7, BIOS_PUTSTR
	call  bios_func_entry


	// 任务2：调用 BIOS 的 bios_sd_read 把内核从 SD 卡读入内存
	//   内核占用的扇区数由 createimage 写在镜像偏移 0x1fc 处
	//   （即内存 0x502001fc，位于 bootblock 自己扇区的尾部），这里用 lh 读出
	//   镜像布局：第 0 扇区 = bootblock，第 1 扇区起 = 内核
	//   加载目标：0x50201000，每个扇区 512 字节
	li    t3, os_size_loc       // t3 = 0x502001fc（扇区数字段地址）
	lh    t2, 0(t3)             // t2 = 内核占用的扇区数
	li    t0, 1                 // t0 = 起始扇区号（SD 卡第 1 扇区）
	li    t1, kernel            // t1 = 目标内存地址

read_loop:
	beqz  t2, read_done         // 扇区数为 0 则不读（防御性判断）

	move  a0, t1                // a0 = 本次读取的目标内存地址
	li    a1, 1                 // a1 = 每次读 1 个扇区
	move  a2, t0                // a2 = 本次读取的扇区号
	li    a7, BIOS_SDREAD       // a7 = 11
	call  bios_func_entry

	addi  t1, t1, 512           // 下一个扇区的目标地址
	addi  t0, t0, 1             // 下一个扇区号
	addi  t2, t2, -1            // 剩余扇区数减一
	j     read_loop

read_done:

	// 任务4/5：把镜像布局参数传给内核（替代内核直接读 0x50200xxx）：
	//   布局：kernel.img = 扇区 0(bootblock) + 1..K(内核) + 1+K(AppInfo, A=1)；
	//         apps.img 与之拼接，从扇区 1+K+A 开始连续排列。
	//   传参约定（head.S 会保存，见 task.h）：
	//     a0 = AppInfo 区起始扇区号 = 1 + K（t0 循环结束时正好等于 1+K）
	//     a1 = 用户程序数目（lh 本扇区偏移 0x1f8）
	//     a2 = apps 区基址 = 1 + K + A（A=1 为 AppInfo 扇区数，即 kernel.img 总扇区数）
	//   这依赖 U-Boot `virtio read 50200000 0 2` 预读前 2 扇区到 0x50200000：
	//   0x502001fc/0x1f8 才在内存里可读（见 log/task3 前置项 3 的隐藏依赖说明）。
	move  a0, t0               // a0 = 1 + K（AppInfo 起始扇区）
	li    t3, os_size_loc - 4  // t3 = 0x502001f8（app 数目字段地址）
	lh    a1, 0(t3)            // a1 = 用户程序数目
	addi  a2, t0, 1            // a2 = 1 + K + A（A=1；apps 区基址）

	// 任务2：跳转到内核入口，启动 UCAS-OS
	jal   kernel

	/************************************************************/
	/* Do not touch this comment. Reserved for future projects. */
	/************************************************************/
// while(1) --> stop here
stop:
	j stop
	nop

.data

msg: .string "It's Dreamerhyh's bootloader...\n\r"

#include <asm.h>
#include <csr.h>

#define KERNEL_STACK		0x50500000

.section ".entry_function","ax"
ENTRY(_start)
  /* Mask all interrupts */
  csrw CSR_SIE, zero
  csrw CSR_SIP, zero

  /* 任务2：清零 BSS 段
   *   起止地址取自 riscv.lds 定义的符号 __bss_start 与 __BSS_END__，
   *   由链接器按本内核实际的段布局赋值，无需硬编码。
   *   镜像里只有已初始化的数据（PT_LOAD 的 p_filesz 部分），
   *   BSS 不在镜像中，必须在这里清零，否则 main 里的 bss_check 会失败。 */
  la    t0, __bss_start
  la    t1, __BSS_END__
clear_loop:
  bge   t0, t1, clear_done
  sd    zero, 0(t0)
  addi  t0, t0, 8
  j     clear_loop
clear_done:

  /* 任务4/5：保存 bootblock 传来的引导参数
   *   a0 = AppInfo 区起始扇区号，a1 = 用户程序数目，a2 = apps 区基址
   *   必须放在清完 BSS 之后：这三个全局量在 .bss 里，先清再存。
   *   此后 main() 从全局量取值，内核不再直接读 0x50200xxx。 */
  la    t2, boot_appinfo_sector
  sd    a0, 0(t2)
  la    t3, boot_appnum
  sd    a1, 0(t3)
  la    t4, boot_apps_base
  sd    a2, 0(t4)

  /* 任务2：建立 C 运行环境
   *   栈顶 0x50500000（KERNEL_STACK），向下生长；
   *   然后跳转到 init/main.c 的 main 函数。 */
  li    sp, KERNEL_STACK
  call  main


loop:
  wfi
  j loop

END(_start)

#ifndef __INCLUDE_TASK_H__
#define __INCLUDE_TASK_H__

#include <type.h>

#define TASK_MEM_BASE    0x52000000
#define TASK_MAXNUM      16
#define TASK_SIZE        0x10000
#define TASK_NAME_LEN    16       /* app 名最长长度（含 '\0'） */

#define SECTOR_SIZE 512
#define NBYTES2SEC(nbytes) (((nbytes) / SECTOR_SIZE) + ((nbytes) % SECTOR_SIZE != 0))

/* Task 4 的镜像布局（紧凑排列，无固定空泡）：
 *
 *   扇区 0                bootblock
 *   扇区 1 .. K           内核，K = NBYTES2SEC(内核实际大小)
 *                          （bootblock 从镜像偏移 0x1fc 读出 K）
 *   扇区 K+1              AppInfo 区（task_info_t 数组，固定 1 扇区，见 APPINFO_SECTORS）
 *   扇区 K+1+APPINFO_SECTORS ...   各 app 按实际大小紧凑排列，app i 紧贴 app i-1
 *
 * 可计算性论证（为什么 AppInfo 一定在 K+1）：
 *   bootblock 唯一能可靠拿到的布局信息是 0x1fc 处的内核扇区数 K，
 *   app 又是紧凑排列的、尾部不可预期，所以"1+K"是 AppInfo 唯一可自定位的位置。
 *   内核侧只要有 K 就能算出 AppInfo 扇区号，进而在启动时读出 task_info_t 数组。
 */
#define APPINFO_SECTORS    NBYTES2SEC(TASK_MAXNUM * sizeof(task_info_t))  /* = 1 */

/* bootblock 经寄存器传给内核的引导参数（Task 4 + Task 5）：
 *   a0 = AppInfo 区起始扇区号
 *   a1 = 用户程序数目（bootblock 从镜像偏移 0x1f8 读出）
 *   a2 = apps 区基址（SD 绝对扇区 = 1+K+A，即 kernel.img 总扇区数）
 * head.S 在清完 BSS 后把这三个值存进下面的全局量，再跳 main。
 * 注意：内核一律不直接读 0x50200xxx（那是 bootblock 扇区，只允许 bootblock 读）。 */
extern uint64_t boot_appinfo_sector;
extern uint64_t boot_appnum;
extern uint64_t boot_apps_base;

/* Task 4 的 task_info_t：按"名称 + 起始扇区 + 扇区数 + 内存入口"设计 */
typedef struct {
    char      name[TASK_NAME_LEN];   /* app 名，与 ELF 文件名一致 */
    uint32_t  start_sector;          /* 该 app 在镜像中的起始扇区号 */
    uint32_t  sector_num;            /* 占用扇区数 = NBYTES2SEC(p_filesz) */
    uint32_t  vaddr_entry;           /* 内存入口地址 */
} task_info_t;

extern task_info_t tasks[TASK_MAXNUM];

#endif```

```c
#include <common.h>
#include <asm.h>
#include <os/kernel.h>
#include <os/task.h>
#include <os/string.h>
#include <os/loader.h>
#include <type.h>

#define VERSION_BUF 50

int version = 2; // version must between 0 and 9
char buf[VERSION_BUF];

// bootblock 经 head.S 传来的引导参数（Task 4/5）：
//   a0 = AppInfo 区起始扇区号，a1 = 用户程序数目，a2 = apps 区基址
//   定义见 task.h；赋值在 head.S 清完 BSS 后完成
uint64_t boot_appinfo_sector;
uint64_t boot_appnum;
uint64_t boot_apps_base;

// Task info array
task_info_t tasks[TASK_MAXNUM];

// 任务3：镜像里记录的用户程序数目（createimage 写在镜像偏移 0x1f8）
int task_num = 0;

static int bss_check(void)
{
    for (int i = 0; i < VERSION_BUF; ++i)
    {
        if (buf[i] != 0)
        {
            return 0;
        }
    }
    return 1;
}

static void init_jmptab(void)
{
    volatile long (*(*jmptab))() = (volatile long (*(*))())KERNEL_JMPTAB_BASE;

    jmptab[CONSOLE_PUTSTR]  = (long (*)())port_write;
    jmptab[CONSOLE_PUTCHAR] = (long (*)())port_write_ch;
    jmptab[CONSOLE_GETCHAR] = (long (*)())port_read_ch;
    jmptab[SD_READ]         = (long (*)())sd_read;
}

static void init_task_info(void)
{
    // 任务4：从 AppInfo 区读出 task_info_t 数组，填进 tasks[]。
    //   位置与数量来自 bootblock 传参（head.S 已存进 boot_appinfo_sector /
    //   boot_appnum），不再直接读 0x50200xxx。
    //   AppInfo 必须先读到本地 512B 对齐缓冲中转，再拷 task_num*sizeof 字节
    //   进 tasks[]，不能整扇区直读 tasks[]（会越界覆写 tasks 后面的 .bss）。
    // NOTE: [p1-task3] 的 taskid 读取已废弃，改按名字加载（见 main 循环）。
    static uint8_t appinfo_buf[SECTOR_SIZE]
        __attribute__((aligned(SECTOR_SIZE)));

    task_num = (int)boot_appnum;

    if (task_num <= 0 || task_num > TASK_MAXNUM) {
        task_num = 0;
        return;
    }

    // 读出 AppInfo 区（固定 APPINFO_SECTORS = 1 个扇区）
    if (bios_sd_read((unsigned)(unsigned long)appinfo_buf, APPINFO_SECTORS,
                     (unsigned)boot_appinfo_sector) != 0) {
        task_num = 0;
        return;
    }

    memcpy((void *)tasks, (const void *)appinfo_buf,
           (size_t)task_num * sizeof(task_info_t));
}

// 任务5：十进制无符号打印（内核无 itoa，除 10 取余、倒序输出）
static void put_uint(unsigned int val)
{
    char tmp[12];
    int i = 0;

    if (val == 0) {
        bios_putchar('0');
        return;
    }
    while (val > 0) {
        tmp[i++] = (char)('0' + val % 10);
        val /= 10;
    }
    while (i > 0)
        bios_putchar(tmp[--i]);
}

// 任务5：十六进制打印（地址用 16 进制更直观）
static void put_hex(unsigned int val)
{
    static const char hex[] = "0123456789abcdef";
    char tmp[9];
    int i = 0;

    bios_putstr("0x");
    if (val == 0) {
        bios_putchar('0');
        return;
    }
    while (val > 0) {
        tmp[i++] = hex[val & 0xf];
        val >>= 4;
    }
    while (i > 0)
        bios_putchar(tmp[--i]);
}

// 任务4：读入一行由 \r 或 \n 结束的程序名。
//   支持退格（BS=0x08 / DEL=0x7f）；名字截断到 TASK_NAME_LEN-1；
//   若某行超长则"整个名字"判非法，清空到行尾后返回 0。
//   \r 与 \n 都立即终止；后续的 \r 或 \n（如 \r\n 的另一半）会以空行被
//   瞬间消费掉，由调用方忽略空行，不会卡住。
static int readline(char *buf, int cap)
{
    int len = 0;

    buf[0] = '\0';
    while (1)
    {
        int ch = bios_getchar();

        if (ch == -1)
            continue;   // 无输入跳过

        if (ch == '\n' || ch == '\r') {   // 双终止符
            buf[len] = '\0';
            return len;
        }

        if (ch == '\b' || ch == 0x7f) {   // 退格：BS 或 DEL
            if (len > 0) {
                len--;
                bios_putchar('\b');        // 回显擦除
                bios_putchar(' ');
                bios_putchar('\b');
            }
            continue;
        }

        if (ch < ' ' || ch > '~')         // 不可打印字符忽略
            continue;

        if (len >= cap - 1) {              // 名字超长：本行作废
            while (1) {
                while ((ch = bios_getchar()) == -1)
                    ;
                if (ch == '\n' || ch == '\r')
                    break;
            }
            buf[0] = '\0';
            return 0;
        }

        bios_putchar(ch);                  // 回显
        buf[len++] = (char)ch;
    }
}

        if (entry == 0)
        {
            bios_putstr("Error: unknown task or load failed!\n\r");
            continue;
        }

        // 跳转到 app 入口；app 的 crt0 会在 app main 返回后跳回内核

        asm volatile(
            "la   ra, 1f\n\t"
            "jalr a0, %0\n\t"
            "1:\n\t"
            :
            : "r"(entry)
            : "ra", "a0", "memory"
        );
    }

    return 0;

#include <assert.h>
#include <elf.h>
#include <errno.h>
#include <stdarg.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define KERNEL_IMAGE_FILE "./kernel.img"
#define APPS_IMAGE_FILE   "./apps.img"
#define ARGS "[--extended] [--vm] <bootblock> <executable-file> ..."

#define SECTOR_SIZE 512
#define BOOT_LOADER_SIG_OFFSET 0x1fe
#define OS_SIZE_LOC (BOOT_LOADER_SIG_OFFSET - 2)
#define BOOT_LOADER_SIG_1 0x55
#define BOOT_LOADER_SIG_2 0xaa

#define NBYTES2SEC(nbytes) (((nbytes) / SECTOR_SIZE) + ((nbytes) % SECTOR_SIZE != 0))

/* 任务4 的 task_info_t，与 include/os/task.h 完全一致（必须保持同步，
 * 大小 16+4+4+4 = 28 字节，4 字节对齐；TASK_MAXNUM * 28 = 448 < 512，
 * 所以 AppInfo 区固定 = 1 扇区）。 */
#define TASK_NAME_LEN 16
#define TASK_MAXNUM   16
typedef struct {
    char     name[TASK_NAME_LEN];
    uint32_t start_sector;
    uint32_t sector_num;
    uint32_t vaddr_entry;
} task_info_t;
static task_info_t taskinfo[TASK_MAXNUM];

/* 镜像信息区（位于第 0 扇区尾部）。用户程序数目 @0x1f8（避开官方签名 @0x1fe）。 */
#define IMG_APP_NUM_LOC (OS_SIZE_LOC - 4)

/* 任务4 镜像布局（紧凑排列，无固定空泡）：
 *
 *   扇区 0                bootblock
 *   扇区 1 .. K           内核，K = NBYTES2SEC(内核实际大小)

 *   扇区 K+2 ..           各 app 按实际大小紧凑排列（app i 紧贴 app i-1）
 *
 * bootblock 从第 0 扇区尾部读出 K 与 app 数目，把 AppInfo 扇区号(1+K)与
 * app 数目经寄存器传给内核（见 arch/riscv/boot/bootblock.S、head.S）。
 */
#define APPINFO_SEC_START(k) ((k) + 1)

/* structure to store command line options */

int main(int argc, char **argv)
{

    options.extended = 0;

/* 预扫描：返回 ELF 中所有 PT_LOAD 段的 p_filesz 之和，并输出入口地址 */
static uint32_t scan_elf(const char *path, uint64_t *entry_out)
{
    FILE *fp = fopen(path, "r");
    Elf64_Ehdr ehdr;
    Elf64_Phdr phdr;
    uint32_t filesz_sum = 0;

    assert(fp != NULL);
    read_ehdr(&ehdr, fp);
    *entry_out = ehdr.e_entry;

    for (int ph = 0; ph < ehdr.e_phnum; ph++) {
        read_phdr(&phdr, fp, ph, ehdr);
        if (phdr.p_type != PT_LOAD)
            continue;
        filesz_sum += get_filesz(phdr);
    }
    fclose(fp);
    return filesz_sum;
}

static void create_image(int nfiles, char *files[])
{
    int tasknum = nfiles - 2;
    int phyaddr = 0;
    FILE *img = NULL, *aimg = NULL;
    Elf64_Ehdr ehdr;
    Elf64_Phdr phdr;

    /* 每个文件的布局信息（顺序：files[0]=bootblock, files[1]=main, files[2..]=app） */
    uint32_t used_bytes[TASK_MAXNUM + 2];
    uint64_t entry_addr[TASK_MAXNUM + 2];

    /* 预扫描一遍文件，得到每个文件实际会写进镜像的字节数 */
    for (int fidx = 0; fidx < nfiles; ++fidx)
        used_bytes[fidx] = scan_elf(files[fidx], &entry_addr[fidx]);

    /* ---- 任务5 双镜像布局（两文件相互解耦） ----
     * kernel.img: 扇区 0 bootblock | 1..K 内核 | K+1 AppInfo(A=1 扇区)
     * apps.img:   各 app 紧凑排列；AppInfo 记录的是 apps.img 内相对扇区号
     * apps 区基址（SD 绝对扇区）= 1 + K + A（= kernel.img 总扇区数）
     */
    int kernel_sectors  = NBYTES2SEC(used_bytes[1]);              /* K */
    int appinfo_sectors = NBYTES2SEC(TASK_MAXNUM * sizeof(task_info_t)); /* A = 1 */
    int apps_base_sector = 1 + kernel_sectors + appinfo_sectors;  /* 传内核 */

    int app_rel_sec[TASK_MAXNUM];   /* apps.img 内相对扇区号 */
    int cur = 0;
    for (int i = 0; i < tasknum; ++i) {
        app_rel_sec[i] = cur;
        cur += NBYTES2SEC(used_bytes[2 + i]);
    }
    int apps_total_sectors = cur;
    int kernelimg_sectors = apps_base_sector;

    printf("===== 任务5 双镜像布局 =====\n");

    /* 填充 taskinfo[]：start_sector 一律为 apps.img 内相对扇区号 */
    for (int i = 0; i < tasknum; ++i) {
        if (strlen(files[2 + i]) >= TASK_NAME_LEN) {
            error("task name '%s' too long (>= %d)\n", files[2 + i],
                  TASK_NAME_LEN);
        }
        strcpy(taskinfo[i].name, files[2 + i]);
        taskinfo[i].start_sector = app_rel_sec[i];
        taskinfo[i].sector_num   = NBYTES2SEC(used_bytes[2 + i]);
        taskinfo[i].vaddr_entry  = (uint32_t)entry_addr[2 + i];
    }
    for (int i = tasknum; i < TASK_MAXNUM; ++i)
        memset(&taskinfo[i], 0, sizeof(task_info_t));

    /* ========== kernel.img ========== */
    img = fopen(KERNEL_IMAGE_FILE, "w");
    assert(img != NULL);

    /* 扇区 0：bootblock */
    FILE *fp = fopen(files[0], "r");
    assert(fp != NULL);
    read_ehdr(&ehdr, fp);
    printf("0x%04lx: %-8s sec %2d .. %2d  (%d bytes, 1 sector)\n",
           ehdr.e_entry, files[0], 0, 0, used_bytes[0]);
    for (int ph = 0; ph < ehdr.e_phnum; ph++) {
        read_phdr(&phdr, fp, ph, ehdr);
        if (phdr.p_type != PT_LOAD)
            continue;
        write_segment(phdr, fp, img, &phyaddr);
    }
    fclose(fp);
    if (phyaddr > SECTOR_SIZE) {
        error("bootblock (%d bytes) exceeds one sector (%d bytes)\n",
              phyaddr, SECTOR_SIZE);
    }
    write_padding(img, &phyaddr, SECTOR_SIZE);

    /* 扇区 1..K：内核 */
    fp = fopen(files[1], "r");
    assert(fp != NULL);
    read_ehdr(&ehdr, fp);
    printf("0x%04lx: %-8s sec %2d .. %2d  (%d bytes, %d sectors)\n",
           ehdr.e_entry, files[1], 1,
           1 + kernel_sectors - 1, used_bytes[1], kernel_sectors);
    for (int ph = 0; ph < ehdr.e_phnum; ph++) {
        read_phdr(&phdr, fp, ph, ehdr);
        if (phdr.p_type != PT_LOAD)
            continue;
        write_segment(phdr, fp, img, &phyaddr);
    }
    fclose(fp);
    write_padding(img, &phyaddr, (1 + kernel_sectors) * SECTOR_SIZE);

    /* 扇区 K+1：AppInfo 区（槽位清零；start_sector 为 apps.img 相对扇区） */
    printf("-APPINFO--- %-8s sec %2d .. %2d  (%d bytes, %d sectors)\n",
           "appinfo", 1 + kernel_sectors,
           1 + kernel_sectors + appinfo_sectors - 1,
           (int)(TASK_MAXNUM * sizeof(task_info_t)), appinfo_sectors);
    write_appinfo(tasknum, (1 + kernel_sectors) * SECTOR_SIZE, img);
    /* fwrite 不经过 phyaddr：同步后补该扇区尾部(448B 后) */
    phyaddr = (1 + kernel_sectors) * SECTOR_SIZE +
              (int)(TASK_MAXNUM * sizeof(task_info_t));
    write_padding(img, &phyaddr, apps_base_sector * SECTOR_SIZE);

    /* ---- 镜像信息字段（0x1f8/0x1fc/0x1fe，写在 kernel.img 第 0 扇区） ---- */
    write_img_info((uint16_t)kernel_sectors, (short)tasknum, img);

    /* kernel.img 大小自检：创作路径上 padding 已保证 phyaddr 恰停在整数个扇区 */
    if (phyaddr != kernelimg_sectors * SECTOR_SIZE) {
        error("kernel.img layout mismatch: computed %d bytes, wrote %d\n",
              kernelimg_sectors * SECTOR_SIZE, phyaddr);
    }
    fclose(img);

    /* ========== apps.img ========== */
    aimg = fopen(APPS_IMAGE_FILE, "w");
    assert(aimg != NULL);
    phyaddr = 0;

    for (int i = 0; i < tasknum; ++i) {
        fp = fopen(files[2 + i], "r");
        assert(fp != NULL);
        read_ehdr(&ehdr, fp);
        printf("0x%04lx: %-8s relsec %2d .. %2d  (%d bytes, %d sectors)\n",
               ehdr.e_entry, files[2 + i], app_rel_sec[i],
               app_rel_sec[i] + taskinfo[i].sector_num - 1,
               used_bytes[2 + i], taskinfo[i].sector_num);
        for (int ph = 0; ph < ehdr.e_phnum; ph++) {
            read_phdr(&phdr, fp, ph, ehdr);
            if (phdr.p_type != PT_LOAD)
                continue;
            write_segment(phdr, fp, aimg, &phyaddr);
        }
        fclose(fp);
        write_padding(aimg, &phyaddr,
                      (app_rel_sec[i] + taskinfo[i].sector_num) * SECTOR_SIZE);
    }

    /* apps.img 大小自检：恰好整扇区且与公式核算一致 */
    if (phyaddr != apps_total_sectors * SECTOR_SIZE)
        error("apps.img layout mismatch: computed %d bytes, wrote %d\n",
              apps_total_sectors * SECTOR_SIZE, phyaddr);
    fclose(aimg);

    printf("\n===== 布局核算 =====\n");
    printf("[layout] kernel.img : %d sectors = 1(boot) + %d(kernel) + %d(appinfo)\n",
           kernelimg_sectors, kernel_sectors, appinfo_sectors);
    printf("[layout] apps.img   : %d sectors = %d(apps)\n",
           apps_total_sectors, apps_total_sectors);
    printf("[layout] apps base (SD 扇区) : %d (= kernel.img 总扇区 = 1+K+A)\n",
           apps_base_sector);
    printf("[layout] app#1 相对扇区 : %d（apps.img 内，无固定空泡）\n",
           tasknum > 0 ? app_rel_sec[0] : -1);
}

static void read_ehdr(Elf64_Ehdr * ehdr, FILE * fp)
{
    int ret;

    ret = fread(ehdr, sizeof(*ehdr), 1, fp);
    assert(ret == 1);
    assert(ehdr->e_ident[EI_MAG1] == 'E');
    assert(ehdr->e_ident[EI_MAG2] == 'L');
    assert(ehdr->e_ident[EI_MAG3] == 'F');
}

static void read_phdr(Elf64_Phdr * phdr, FILE * fp, int ph,
                      Elf64_Ehdr ehdr)
{
    int ret;

    fseek(fp, ehdr.e_phoff + ph * ehdr.e_phentsize, SEEK_SET);
    ret = fread(phdr, sizeof(*phdr), 1, fp);
    assert(ret == 1);
    if (options.extended == 1) {
        printf("\tsegment %d\n", ph);
        printf("\t\toffset 0x%04lx", phdr->p_offset);
        printf("\t\tvaddr 0x%04lx\n", phdr->p_vaddr);
        printf("\t\tfilesz 0x%04lx", phdr->p_filesz);
        printf("\t\tmemsz 0x%04lx\n", phdr->p_memsz);
    }
}

static uint32_t get_filesz(Elf64_Phdr phdr)
{
    return phdr.p_filesz;
}

static void write_segment(Elf64_Phdr phdr, FILE *fp, FILE *img, int *phyaddr)
{
    if (phdr.p_memsz != 0 && phdr.p_type == PT_LOAD) {
        /* write the segment itself */
        /* NOTE: expansion of .bss should be done by kernel or runtime env! */
        if (options.extended == 1) {
            printf("\t\twriting 0x%04lx bytes\n", phdr.p_filesz);
        }
        fseek(fp, phdr.p_offset, SEEK_SET);
        while (phdr.p_filesz-- > 0) {
            fputc(fgetc(fp), img);
            (*phyaddr)++;
        }
    }
}

static void write_padding(FILE *img, int *phyaddr, int new_phyaddr)
{
    if (options.extended == 1 && *phyaddr < new_phyaddr) {
        printf("\t\twrite 0x%04x bytes for padding\n", new_phyaddr - *phyaddr);
    }

    while (*phyaddr < new_phyaddr) {
        fputc(0, img);
        (*phyaddr)++;
    }
}

/* 任务4：把 task_info_t 数组写到 AppInfo 扇区。
 *   整个数组（TASK_MAXNUM 个槽位）都写，未用的槽位在调用前已清 0。 */
static void write_appinfo(int appnum, int appinfo_offset, FILE *img)
{
    fseek(img, appinfo_offset, SEEK_SET);
    fwrite(taskinfo, sizeof(task_info_t), TASK_MAXNUM, img);
    if (appnum > 0) {
        printf("[appinfo] %d task(s) @ 0x%04x\n", appnum, appinfo_offset);
    }
}

static void write_img_info(uint16_t ksectors, short tasknum, FILE * img)
{
    /* 任务4：内核扇区数写"实际占用"（紧凑布局下 bootblock 精确读内核），
     * app 数目 @0x1f8，官方签名 @0x1fe 不动。 */
    fseek(img, IMG_APP_NUM_LOC, SEEK_SET);
    fputc(tasknum & 0xff, img);
    fputc((tasknum >> 8) & 0xff, img);

    fseek(img, OS_SIZE_LOC, SEEK_SET);
    fputc(ksectors & 0xff, img);
    fputc((ksectors >> 8) & 0xff, img);

    fseek(img, BOOT_LOADER_SIG_OFFSET, SEEK_SET);
    fputc(BOOT_LOADER_SIG_1, img);
    fputc(BOOT_LOADER_SIG_2, img);

    printf("\n[img info] kernel sectors : %d @ 0x%03x\n", ksectors, OS_SIZE_LOC);
    printf("[img info] app number     : %d @ 0x%03x\n", tasknum, IMG_APP_NUM_LOC);
    printf("[img info] boot signature : 0x%02x 0x%02x @ 0x%03x\n",
           BOOT_LOADER_SIG_1, BOOT_LOADER_SIG_2, BOOT_LOADER_SIG_OFFSET);
}

/* print an error message and exit */
static void error(char *fmt, ...)
{
    va_list args;

    va_start(args, fmt);
    vfprintf(stderr, fmt, args);
    va_end(args);
    if (errno != 0) {
        perror(NULL);
    }
    exit(EXIT_FAILURE);
}```

## 8. 学习问答集（按任务整理）

> 本附录收录学习过程中实际提出的问题与答案，按"通用 → 任务 1 → … → 任务 5"排列。
> 每个答案都尽量给出**可验证的证据**（源码位置 / 反汇编 / 实测数值），便于汇报时引用。

### 8.0 通用基础

**Q1：扇区是什么？**
磁盘的最小读写单位，本课规定 **1 扇区 = 512 字节**。不能只读"半格"，只能整格整格地读写。
这是 bootblock 被限制在 512 字节以内的根源，也是 BIOS 的 `sd_read` 只能按"块"为单位搬运的原因。

**Q2：QEMU 是什么？它模拟的内存是我们电脑的内存吗？**
是**整机模拟器**：CPU、内存、串口、磁盘都由它用软件扮演。命令行 `-m 256M` 表示它向 Linux 要 256MB，
"扮演"成这台假电脑的内存条——所以内存**物理上是你真电脑的**，但组织方式由 QEMU 扮演。
`-drive file=build/image` 则是把镜像文件扮演成一块磁盘。

**Q3："从 0 设计操作系统"——哪些是给定的，哪些要自己写？**

| 层次 | 谁提供 | 要写吗 |
|---|---|---|
| 真电脑 + Linux + QEMU（假电脑） | 环境自带 | ❌ |
| u-boot 固件（= BIOS：读盘/打印/读键） | 环境自带（仓库外 `~/OSLab-RISC-V/u-boot/u-boot`） | ❌ |
| 工具链 + Makefile + `riscv.lds` + start-code 骨架 | 课程发放 | ❌ |
| bootblock、内核入口、任务加载、createimage、crt0 | **我们** | ✅ |

"从 0"的准确含义：**脚下没有操作系统垫底**（没有 Linux、没有 libc），我们写的就是那层"世界本身"。

**Q4：u-boot 在不在仓库里？它和我们的代码是什么关系？**
**不在仓库里**，是环境提供的预编译固件（`~/OSLab-RISC-V/u-boot/u-boot`）。
仓库里与 BIOS 相关的只有两样："号码表"`arch/riscv/include/asm/biosdef.h` 和"打电话的示范"
`arch/riscv/bios/common.c`——**没有一行是 BIOS 的实现**。

**Q5：镜像是什么？是磁盘吗？创建镜像时数据从哪来？**
镜像是**一块磁盘内容的一比一字节副本**（一个普通文件）。QEMU 把它挂成虚拟磁盘，"读磁盘"就是读这个文件。
创建时的数据来自 `build/` 里的 ELF：`createimage` 解析 ELF，抠出 PT_LOAD 段里真正要执行的字节，
按扇区对齐拼成镜像。所以镜像 = 多个 ELF 的精华 + 对齐补零 + 约定字段（`0x1fc` 等）。

**Q6：u-boot 和我们的代码在同一片内存里吗？M 态比"内核态"权限高吗？**

```
0x50000000 ┌────────────────┐
           │ u-boot（M 态）  │ ← 比我们"靠前"的 2MB
0x501FFFFF └────────────────┘
0x50200000 ┌────────────────┐
           │ bootblock      │ ← 我们操作系统的起点
0x50201000 │ 内核           │
           └────────────────┘
```
同一片物理内存，分段居住。权限等级 **M（机器态）> S（管理员态）> U（用户态）**：
u-boot 在 M 态，我们的代码在 **S 态**，`ecall` 就是从 S 态向 M 态"打报告"。
⚠️ 注意：**"不能碰 u-boot 的内存"没有硬件屏障**——`loadboot` 把 PMP 设成整个地址空间 RWX（Phase 0 §3 实测）。
不去碰是**约定 + 后果**（u-boot 还要继续当服务台）。

---

### 8.1 任务一

**Q1：bootblock 是代码还是什么？是给定的还是自己写的？**
它是**真正会被 CPU 执行的第一段代码**。文件骨架 `arch/riscv/boot/bootblock.S` 是 start-code 发放的
（里面是 `TODO: [p1-task1]` 路标），**具体代码是我们填的**：三行汇编 + 一个字符串。

**Q2：512 字节限制是哪来的？不足 512 会怎样？**
两个来源：① 约定"第 0 扇区 = bootblock"，打包工具只把它补到正好 1 扇区；
② **更硬**的约束：扇区尾部 `0x1fc`（内核扇区数）与 `0x1fe`（`55aa` 签名）共占 4 字节，
所以 bootblock 代码+数据**实际不能越过 `0x1fc`，有效上限 508 字节**。不足部分由 createimage 补 0。

**Q3：为什么要传 a0/a7？功能号在哪定义？**
约定：**参数进 a0、a1…，功能号进 a7**，然后跳 `0x50150000`。
功能号的名字定义在 `arch/riscv/include/asm/biosdef.h`（5 行宏，`BIOS_PUTSTR=9` 等），铁律规定不许改。

**Q4：BIOS 搬的"我们的代码"在仓库里指哪一块？**
指 **`build/image` 的前 512 字节**。它是这样来的：
`arch/riscv/boot/bootblock.S` --make elf--> `build/bootblock`(ELF) --createimage--> `build/image` 第 0 扇区。

**Q5：`55 aa` 签名干什么用？**
引导扇区末尾的"合法标记"（x86 时代的祖传规矩）。作用是用极小代价判断"这段是不是引导代码"
（随机数据碰巧撞上 `55 aa` 的概率仅 1/65536）。本课的 u-boot 其实不检查它，但既然官方工具在那儿刻字，
自建工具就沿用，保持镜像格式的标准性。

**Q6：结尾为什么必须 `j stop` 自旋？**
打印完不能"往下掉"——后面是 `.data` 里的字符串字节，CPU 会把它们当指令解读，当场跑飞。
原地死循环是最安全的收尾（也是验收断言"标志串只打印一次"的底层依据）。

### 8.2 任务二

**Q1：内核一开始在哪？一定要全部搬进内存才能用吗？**
内核以"抠出来的指令字节"形态躺在**镜像的第 1~K 扇区**（磁盘上）。
**必须搬进内存才能执行**：① CPU 只能从内存取指令（块设备没有地址线，不能"原地执行"）；
② 没有文件系统、没有缺页机制。但"全部"有精确含义：**只搬 PT_LOAD 的字节，BSS 不在盘上**，
由 `head.S` 在内存里现清。

**Q2：`jal kernel` 为什么就是 head.S？**
因为 `kernel` 只是 `.equ kernel, 0x50201000` 这个**地址外号**，与文件名无关。而"0x50201000 上恰好是
head.S"是 **Makefile + 链接脚本做的媒**：
`KERNEL_ENTRYPOINT=0x50201000` → `riscv.lds` 的 `. = TEXT_START;` 且 `*(.entry_function)` 排最前 →
head.S 的 `ENTRY(_start)` 正好落在该地址。**两个数字对上，程序就连上了。**

**Q3：main.c 里哪些是原始给的，哪些是任务 2/3/4/5 加的？**

| 内容 | 来自 |
|---|---|
| 欢迎语打印、`bss_check()`、**`init_jmptab()`**、`buf`/`version` 定义、`while(1){wfi;}` | **start-code 原始**（任务 0） |
| 键盘回显循环（`-1` 跳过） | 任务 2（后被覆盖） |
| `read_taskid()`、按编号分发、`la ra,1f` 跳转 | 任务 3（后被覆盖） |
| `boot_appinfo_sector`/`boot_appnum`、`init_task_info()` 实现、`readline()`、按名加载 | 任务 4 |
| `boot_apps_base`、`put_uint`/`put_hex`/`show_task_list`、`list` 命令 | 任务 5 |

**Q4：`buf` 是什么？"version: 2"怎么算出来的？它怎么证明 BSS 被清了？**
`buf[50]` 是**双面道具**：先当 BSS 的"证人"（它是 `.bss` 里没人初始化的变量，应当全 0），
`bss_check()` 读它全 0 就返回 1，否则 0；检查完再被**覆写**成打印缓冲。
`version` 与体检**无关**：`output_val[1] = version + '0'` = `2 + 48` = `'2'`，永远打 2。
**真正的证据是 `t`/`f`**，它来自 `check = bss_check()`。

**Q5：`buf` 是全部 BSS 吗？**
**不是**，它只是 50 字节的**抽样证人**（BSS 全部 1512 字节，buf 占 3.3%）。实测布局：

```
0x50201bd8 __bss_start / task_num
0x50201be0 boot_apps_base
0x50201be8 boot_appnum
0x50201bf0 boot_appinfo_sector
0x50201c00 buf            ← 证人（50 字节）
0x50201e00 appinfo_buf.1  (512B 对齐缓冲)
0x50202000 tasks          (16×28 名册)
0x502021c0 __BSS_END__
```
抽样够用的原因：清 BSS 是一整段循环，要么整片生效要么整片失效，不存在"优雅地只坏中间一块"。
（历史细节：任务二时 buf 恰好就是 `__bss_start` 本身，后来新全局量插到了它前面。）

**Q6：为什么现在 main.c 里的回显循环不像任务二那版了？**
被任务 3、4、5 连改三代：任务 2 是"读一键回显一键"；任务 3 换成"读数字编号 + 分发"；
任务 4/5 换成"读一行名字 + 查名册 + `list` 命令"。但**有一行从第一代活到现在**：
`if (ch == -1) continue;`（现在藏在 `readline()` 里）——"宁可跳过，绝不把 -1 当字符打出去"。

**Q7：`common.c` 的 `call_bios` 是什么？是内联汇编吗？**
**不是汇编**，是**利用 RISC-V 调用约定写的纯 C**：把 `0x50150000` 强制转成函数指针，
声明成"8 个参数"，然后把功能号放在**第 8 个实参**位置——编译器就会按 ABI 把它放进 **a7**，
其余参数正好落在 a0~a4。中间两个 `IGNORE` 是占位（跳过 a5/a6）。
**本质是"摆好寄存器 + 跳过去 + 跳回来"，伪装成函数调用。**

**Q8：为什么 `call_bios` 形参 6 个、调用时却传 8 个、顺序还不一样？**
两层不同的东西：**形参表是给人读的**（功能号在前）；**实参顺序是给硬件摆的**（功能号必须在 a7 = 第 8 位）。
`call_bios` 就是这两者的翻译官。

**Q9：`common.c` 和 `kernel.h` 是递进还是并列？**
**递进**（一上一下）：

```
main.c → bios_putstr (kernel.h，static inline) → call_jmptab 查跳转表 0x51ffff00
       → port_write (common.c，住在 0x502015ac) → call_bios → 0x50150000 → u-boot
```
**kernel.h 只认识跳转表；common.c 才认识 BIOS 的门。**
顺序：**先查表，后敲 BIOS**（call_bios 是最里层、最后一步）。

**Q10：把数字当函数指针，和普通 C 调用差在哪？**
差的只有"地址从哪来"：普通 C 用**符号名**（链接器自动填地址）；这里因为 u-boot 是与我们独立编译的
程序、我们的 ELF 里没有代表它的符号，所以只能**写死数字 + 强制类型转换**。
一旦地址确定，调用行为完全一样。

**Q11：跳转表那句声明太绕，怎么读？**
`volatile long (*(*jmptab))()` 从里往外读：jmptab 是指针 → 指向"函数指针数组"。等价可读版：

```c
typedef long (*bios_fn_t)();
volatile bios_fn_t *jmptab = (volatile bios_fn_t *)KERNEL_JMPTAB_BASE;
jmptab[CONSOLE_PUTSTR] = port_write;
```

内存分布：

```
0x51ffff00 槽0: port_write 的地址 (0x502015ac)
0x51ffff08 槽1: port_write_ch 的地址
0x51ffff10 槽2: port_read_ch 的地址
0x51ffff18 槽3: sd_read 的地址
0x51ffff20 槽4: 未填（sd_write 槽，至今是 0 —— 潜坑）
```
**表里存的是"代码在哪"，表本身是数据，不是代码。**

**Q12：`common.h` 和 `common.c` 谁干活？`common.c` 在 `0x51ffff00` 吗？**
`arch/riscv/include/common.h` 只有**声明**（函数原型），是"菜单"；
`arch/riscv/bios/common.c` 才是**实现**（干活的）。
**`common.c` 不在 `0x51ffff00`**：它的代码住在内核区 **0x50201598~0x5020160a**；
`0x51ffff00` 是跳转表（数据），**表里记着 common.c 那几个函数的地址**。

**Q13：`port_write` 为什么在 `0x502015ac`？**
它是 common.c 里的函数，被编译进内核；链接器按 `TEXT_START=0x50201000` 往后排，恰好排到 `0x502015ac`。
Makefile 只给了"起点"一个数字，**具体每个函数的地址是链接器分配的**（同文件的函数基本按书写顺序递增排列，
但这是工具链的自由，程序里只该认符号名，不该依赖数字）。
另外：`call_bios` 在最终产物里**没有自己的地址**——被 `-O2` 内联进了全部调用者。

**Q14：BIOS 的门 `0x50150000` 上是什么？**
u-boot 写在那里的**两条指令**：`ecall`（报信）+ `ret`（送客）。
它不是"某个函数"，而是**陷阱门/中转站**：进去 → ecall 陷入 M 态 → u-boot 按 a7 分发 → 干完 mret 回来 →
`ret` 用 ra 送回调用者。

**Q15：从陷阱门返回，是直接回到 `main.c` 的下一条吗？**
**看调用链，两种都真实存在**（同一份代码里并存）：

- **样本 A（`bios_putstr`，尾跳穿透）**：
  `main --jal--> bios_putstr --jr--> port_write --jr--> 0x50150000`。
  两跳都是 `jr`（不写 ra），所以 **BIOS 的 ret 直接回到 main 的调用点下一条**。
- **样本 B（`sd_read`，带链接 + 现场保存）**：`sd_read` 开了栈帧、`sd ra,8(sp)` 保存了 ra，
  然后用 **`jalr t1`（带链接）** 跳 BIOS，返回后还要 `sext.w` 加工返回值，最后 `ret` 回 main。
  所以 **BIOS 先回到 `sd_read`，`sd_read` 再回 main**。

**规律：返回目标永远由 `ra` 决定；谁在跳之前写了 ra，`ret` 就回谁那儿。**
编译器会自行决定用"尾跳穿透"还是"带链接跳 + 现场保存"。

### 8.3 任务三

**Q1：入口地址为什么不从镜像读 `e_entry`，而用窗口基址？**
三条依据：① `riscv.lds` 里 `ENTRY(_start)` 且 `.text` 首段是 `*(.entry_function)`；
② `crt0.S` 的 `_start` 定义在该段，排在 app 代码之前；③ Makefile 用
`-Wl,--defsym=TEXT_START=<窗口基址>` 给每个 app 单独赋值。
实测四个 app 的 `e_entry` 与窗口基址完全一致（`0x52000000/0x52010000/0x52020000/0x52030000`）。
任务 4 设计 `task_info_t` 后，`e_entry` 才被正式存进名册。

**Q2：crt0 为什么要存内核的 sp/ra？为什么必须放 `.data` 槽？**
因为 app 的 main 会随意使用 a0 等寄存器（不能寄存），而 `s0-s11` 是被调用者按 ABI **必须保全**的
（内核 main 里活着的 s 寄存器不能被破坏）——所以只能存进内存的固定槽。
**槽必须在 `.data` 而不是 `.bss`**：否则紧接着的 BSS 清零循环会把自己刚存的值抹掉。
本函数全程只用 t/a 寄存器，不碰 s 寄存器。

**Q3：app 的 BSS 边界信息从哪来？**
同一份 `riscv.lds` 在**每个 app 自己的布局**中给 `__bss_start`/`__BSS_END__` 赋值；
`crt0.S` 引用这两个符号名，链接时就解析成"当前这个 app"的边界。不需要 bootblock 传参。
（注意必须用 `sb` 逐字节清：`auipc` 的 `__bss_start = 0x52010284` 非 8 对齐、bss 仅 4 字节，
用 `sd` 会越界踩坏后面的内存。）

**Q4：app 栈为什么是共享的 `0x50400000`？**
最初提案"贴内核栈放 `0x504fffff`"被驳回——与内核栈 `0x50500000` 只差 1 字节，
两个向下生长的栈顶部必然交叠。改为共享栈顶 `0x50400000`：内核栈独享 0x50400000–0x50500000，
app 栈向下还有约 2MB 空地。所有 app 串行执行（一个退场下一个才上场），共享一条栈是安全的。

**Q5：app 怎么打印？**
查**同一张跳转表** `0x51ffff00`：app 的 `tiny_libc/include/kernel.h` 里 `bios_putstr` = "查表 → 调到内核的
`port_write`" → 再由内核去敲 BIOS。**app 不认识 BIOS，只认识内核给它的电话簿**。
两份 kernel.h 不同：内核侧 5 个槽（含 SD_READ/SD_WRITE），app 侧只有 3 个（不让 app 直接操作磁盘）。

### 8.4 任务四

**Q1：AppInfo 名册在哪？"`K+1`"是什么意思？**
名册是**夹在内核与 app 队列之间的一个专用扇区**，位于扇区号 `1+K`（K = 内核实际占用扇区数）。
"K+1"就是"数格子的结果"：内核从第 1 格起占 K 格，下一个空位就是 `1+K` 号格。
**可计算性论证**：bootblock 唯一能可靠拿到的布局信息就是 `0x1fc` 里的 K，
所以"1+K"是名册**唯一能自定位**的位置——内核占 3 格名册自动滑到 4 号格，占 9 格滑到 10 号格，无需额外信息。

**Q2：名册为什么不放在 apps.img 里？**
如果名册放在 apps.img，**内核启动时就非得读应用区不可**——这与任务书
"在内核启动的过程中，只把内核镜像从 SD 卡读取到内存中"**直接冲突**。所以名册必须在 kernel.img 里。

**Q3：为什么读名册要先经 512B 缓冲，不能直接读进 `tasks[]`？**
名册扇区是 **512 字节**，但有用部分只有 **448 字节**（16 槽 × 28 字节）。
若让 BIOS 整扇区直读 `tasks[]`，多出的 64 字节会**越界覆写 tasks 后面的 .bss**。
正确做法：先读进一块 **512 字节对齐的缓冲**，再只拷 448 字节进 `tasks[]`。

**Q4：任务书要求"加 -O2"，为什么汇报里说没有对比基线？**
因为 start-code 的 `CFLAGS` **一开始就带 `-O2`**（Phase 0 勘察确认），不是任务四才加的，
也不存在"任务三未优化"的基线可对比。所以如实报告"已生效"，**不伪造对比数据**。

**Q5：head.S 的"对齐复查"是什么？**
`head.S` 清 BSS 用 `sd`（一次 8 字节），**隐含假设 `__bss_start` 是 8 的倍数**。
任务二记录了这个假设，任务四代码结构变化后**复查**：`__bss_start = 0x50201bd8`，8 对齐成立。
（对比：app 那边的同类循环因为边界非对齐，只能用 `sb`。）

### 8.5 任务五

**Q1：为什么要把内核和应用拆成两个镜像？为什么不能挂两块盘？**
任务书原文："模拟文件系统的分区管理……在内核启动的过程中，只把内核镜像从 SD 卡读取到内存中。
用户程序在要执行的时候再动态的从 SD 卡中加载到内存地址执行。"
好处是"app 很多的时候也不影响内核启动的读卡时间"。
**但不能真的挂两块盘**：BIOS 的 `sd_read` 在 u-boot 侧硬编码只读**第 0 块设备**
（`blk_read_devnum(if_type_virtio, 0, ...)`，Phase 0 §3 实测）。
所以落地方案是：**单个 SD 卡镜像内划分"内核区 + 应用区"**——前半截是 kernel.img，后半截是 apps.img，
等价于"把两个文件分别写入一块卡上的连续扇区"。

**Q2：名册里的 `start_sector` 为什么写成"apps.img 内的相对扇区号"？**
为了**让两个镜像真正解耦**：apps.img 增删改内容时，kernel.img 一个字都不用动。
运行时用 `boot_apps_base + start_sector` 换算成 SD 绝对扇区，其中 `boot_apps_base = 1+K+A`
（= kernel.img 的总扇区数）由 bootblock 经 a2 传给内核。

**Q3：`cat kernel.img apps.img > image` 为什么等价于"分别写入 SD 卡"？**
因为两个文件都是**整扇区长度**（kernel.img = `(1+K+A)×512`，apps.img = `Σapp×512`），
首尾相接处正好落在扇区边界上。拼接后的 image 与"把两个文件分别 dd 到一块卡的连续区域"
**字节级完全一致**。任务书说"修改 Makefile，将两个文件都写入 SD 卡"，cat 就是这句话在 QEMU 下的等价物。

**Q4：怎么证明"启动阶段没有读应用数据"？**
**代码路径审查**（bootblock / head.S / 内核 init 的全部读盘调用）：

```
u-boot 预读    : 扇区 0..1
bootblock 读   : 扇区 1..K（内核）
内核 init 读   : 扇区 1+K（名册 AppInfo）
────────────────────────────────────────
合计 0 .. 1+K+A-1  ⊂ kernel.img（apps 区零读取）
```
应用数据只在**输入名字启动某个 app 时**才被读取（`kernel/loader/loader.c` 里唯一的一处调用）。

**Q5：`list` 命令是什么？**
一个信息显示命令（slide 25 的"可选项"）：输入 `list` 后打印名册里已有的内容——
名字 / 起始（相对扇区）/ 大小 / 入口地址，用来直观确认"内核已经拿到应用信息、但一个都没加载"。
